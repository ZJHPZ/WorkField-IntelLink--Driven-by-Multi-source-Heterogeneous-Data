# 本地前后端文件上传至智能体 - 方案文档

## 1. 方案概述

### 1.1 整体架构

```
┌─────────────┐    ┌─────────────┐    ┌─────────────┐    ┌─────────────┐
│   前端页面   │───▶│   后端服务   │───▶│   文件存储   │    │   智能体    │
│  (上传组件)  │    │  (API服务)   │    │  (本地/S3)  │◀───│  (Agent)    │
└─────────────┘    └─────────────┘    └─────────────┘    └─────────────┘
      │                  │                                        │
      │  1.选择文件       │  2.上传文件                            │
      │─────────────────▶│                                        │
      │                  │  3.保存文件                             │
      │                  │────────────────▶                       │
      │                  │                                        │
      │                  │  4.返回文件路径                         │
      │◀─────────────────│                                        │
      │                  │                                        │
      │  5.发送消息(含路径)│  6.查询文件内容                        │
      │─────────────────▶│─────────────────▶                       │
      │                  │                                        │
      │                  │  7.Agent分析结果                        │
      │◀─────────────────│◀───────────────────────────────────────│
      │                  │                                        │
      │  8.展示分析结果   │                                        │
      └─────────────┘                                              │
```

### 1.2 方案选择

| 方案 | 适用场景 | 优点 | 缺点 |
|------|----------|------|------|
| **方案A：本地存储** | 开发测试、小规模使用 | 简单、无需额外服务 | 文件管理不便、不适合生产 |
| **方案B：对象存储** | 生产环境、大规模使用 | 可扩展、CDN加速 | 需要配置云服务 |
| **方案C：Base64内嵌** | 小文件、简单场景 | 无需存储 | 有大小限制、增加请求体积 |

**推荐：方案A（开发） + 方案B（生产）**

```python
# backend/app/storage/local_storage.py
import os
import uuid
import aiofiles
from datetime import datetime
from typing import Optional, Tuple

class LocalStorage:
    """本地文件存储服务"""
    
    def __init__(self, base_dir: str = "uploads/files"):
        self.base_dir = base_dir
        self._ensure_dir()
    
    def _ensure_dir(self):
        """确保目录存在"""
        os.makedirs(self.base_dir, exist_ok=True)
    
    def _generate_filename(self, original_filename: str) -> str:
        """生成唯一文件名"""
        ext = os.path.splitext(original_filename)[1].lower()
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        unique_id = uuid.uuid4().hex[:8]
        return f"{timestamp}_{unique_id}{ext}"
    
    async def save_file(self, file_content: bytes, original_filename: str) -> Tuple[str, str]:
        """
        保存文件
        
        Returns:
            (file_path, file_url)
        """
        filename = self._generate_filename(original_filename)
        file_path = os.path.join(self.base_dir, filename)
        
        async with aiofiles.open(file_path, 'wb') as f:
            await f.write(file_content)
        
        # 返回相对路径（供Agent使用）
        relative_path = file_path
        return relative_path, filename
    
    def get_full_path(self, relative_path: str) -> str:
        """获取完整文件路径"""
        if os.path.isabs(relative_path):
            return relative_path
        return os.path.join(os.getcwd(), relative_path)
    
    def delete_file(self, file_path: str) -> bool:
        """删除文件"""
        try:
            full_path = self.get_full_path(file_path)
            if os.path.exists(full_path):
                os.remove(full_path)
                return True
            return False
        except Exception:
            return False


# 全局实例
storage = LocalStorage()
```

```python
# backend/app/api/upload.py
from fastapi import APIRouter, UploadFile, File, HTTPException
from pydantic import BaseModel
from typing import Optional
import logging

from ..storage.local_storage import storage

logger = logging.getLogger(__name__)
router = APIRouter(prefix="/api/upload", tags=["文件上传"])

# 支持的文件类型
ALLOWED_EXTENSIONS = {'.txt', '.docx', '.pdf', '.doc', '.md'}
MAX_FILE_SIZE = 50 * 1024 * 1024  # 50MB


class UploadResponse(BaseModel):
    success: bool
    file_path: Optional[str] = None
    filename: Optional[str] = None
    file_size: Optional[int] = None
    message: str


@router.post("/file", response_model=UploadResponse)
async def upload_file(file: UploadFile = File(...)):
    """
    上传文件接口
    
    支持格式：.txt, .docx, .pdf, .doc, .md
    最大文件大小：50MB
    """
    # 1. 检查文件扩展名
    filename = file.filename or "unknown"
    ext = '.' + filename.rsplit('.', 1)[-1].lower() if '.' in filename else ''
    
    if ext not in ALLOWED_EXTENSIONS:
        raise HTTPException(
            status_code=400,
            detail=f"不支持的文件格式 {ext}，支持的格式：{', '.join(ALLOWED_EXTENSIONS)}"
        )
    
    # 2. 读取文件内容
    try:
        content = await file.read()
    except Exception as e:
        logger.error(f"读取文件失败: {e}")
        raise HTTPException(status_code=500, detail="文件读取失败")
    
    # 3. 检查文件大小
    file_size = len(content)
    if file_size > MAX_FILE_SIZE:
        raise HTTPException(
            status_code=400,
            detail=f"文件过大，最大支持 {MAX_FILE_SIZE // (1024*1024)}MB"
        )
    
    # 4. 保存文件
    try:
        file_path, saved_filename = await storage.save_file(content, filename)
        logger.info(f"文件保存成功: {file_path}")
    except Exception as e:
        logger.error(f"保存文件失败: {e}")
        raise HTTPException(status_code=500, detail="文件保存失败")
    
    return UploadResponse(
        success=True,
        file_path=file_path,
        filename=saved_filename,
        file_size=file_size,
        message="文件上传成功"
    )


@router.post("/image", response_model=UploadResponse)
async def upload_image(file: UploadFile = File(...)):
    """
    上传图片接口（用于多模态分析）
    
    支持格式：.jpg, .jpeg, .png, .gif, .webp
    最大文件大小：10MB
    """
    ALLOWED_IMAGE_EXTENSIONS = {'.jpg', '.jpeg', '.png', '.gif', '.webp'}
    MAX_IMAGE_SIZE = 10 * 1024 * 1024  # 10MB
    
    ext = '.' + filename.rsplit('.', 1)[-1].lower() if '.' in (filename := file.filename or '') else ''
    
    if ext not in ALLOWED_IMAGE_EXTENSIONS:
        raise HTTPException(
            status_code=400,
            detail=f"不支持的图片格式 {ext}"
        )
    
    content = await file.read()
    if len(content) > MAX_IMAGE_SIZE:
        raise HTTPException(status_code=400, detail="图片过大，最大支持10MB")
    
    file_path, saved_filename = await storage.save_file(content, filename or "image.png")
    
    return UploadResponse(
        success=True,
        file_path=file_path,
        filename=saved_filename,
        file_size=len(content),
        message="图片上传成功"
    )
```

#### 2.2.4 Agent 调用服务

```python
# backend/app/services/agent_service.py
import httpx
import os
from typing import Optional, Dict, Any

class AgentService:
    """Agent 调用服务"""
    
    def __init__(self, agent_url: str = None):
        # Agent 服务地址（由部署环境提供）
        self.agent_url = agent_url or os.getenv("AGENT_URL", "http://localhost:9000")
        self.timeout = 120.0  # 超时时间（秒）
    
    async def analyze_file(self, file_path: str, query: str = None) -> Dict[str, Any]:
        """
        调用 Agent 分析文件
        
        Args:
            file_path: 文件绝对路径
            query: 附加查询指令
        
        Returns:
            Agent 返回的分析结果
        """
        # 构建消息
        if query:
            message = f"{query}\n\n请分析这个文件，路径是：{file_path}"
        else:
            message = f"请分析我上传的学习文件，路径是：{file_path}"
        
        async with httpx.AsyncClient(timeout=self.timeout) as client:
            response = await client.post(
                f"{self.agent_url}/api/chat",
                json={
                    "message": message,
                    "stream": False
                }
            )
            response.raise_for_status()
            return response.json()
    
    async def analyze_image(self, image_path: str, query: str = None) -> Dict[str, Any]:
        """
        调用 Agent 分析图片
        
        Args:
            image_path: 图片绝对路径
            query: 附加查询指令
        """
        if query:
            message = f"{query}\n\n请分析这张图片，路径是：{image_path}"
        else:
            message = f"请描述这张图片的内容：{image_path}"
        
        async with httpx.AsyncClient(timeout=self.timeout) as client:
            response = await client.post(
                f"{self.agent_url}/api/chat",
                json={
                    "message": message,
                    "stream": False
                }
            )
            response.raise_for_status()
            return response.json()


# 全局实例
```

```vue
import { ElMessage } from 'element-plus'
import { UploadFilled, Document } from '@element-plus/icons-vue'

const props = defineProps({
  // 上传接口地址

const uploading = ref(false)

const beforeUpload = (file) => {
  const allowedTypes = ['.txt', '.docx', '.pdf', '.doc', '.md']
  const ext = '.' + file.name.split('.').pop().toLowerCase()
  
  if (!allowedTypes.includes(ext)) {
    ElMessage.error('不支持的文件格式')
    return false

  if (file.size > maxSize) {
    ElMessage.error('文件大小不能超过 50MB')
    return false
  }
  
  uploading.value = true
  percentage.value = 0
  return true
}

  percentage.value = Math.round(event.percent || 0)
}

// 上传成功
const handleSuccess = (response, file) => {
  uploading.value = false
  percentage.value = 100

      emit('analyze-complete', response.file_path)
    }
  } else {
    ElMessage.error(response.message || '上传失败')
    emit('upload-error', response)
  }
}

// 上传失败
const handleError = (err, file) => {
  uploading.value = false
  ElMessage.error('文件上传失败')
  emit('upload-error', { error: err.message })
}

const formatFileSize = (bytes) => {
  if (!bytes) return '0 B'
  const k = 1024
  const sizes = ['B', 'KB', 'MB', 'GB']
  const i = Math.floor(Math.log(bytes) / Math.log(k))
  return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i]
}

  uploadRef.value?.submit()
}

// 清空已上传文件

  uploadRef.value?.clearFiles()
  uploadedFile.value = null
  percentage.value = 0
}

defineExpose({ upload, clear })
```

```python
# backend/app/storage/oss_storage.py
import oss2
import uuid
from datetime import datetime
from typing import Tuple

class OSSStorage:
    """阿里云 OSS / AWS S3 存储服务"""
    
    def __init__(self):
        # 配置（建议使用环境变量）
        self.access_key_id = os.getenv('OSS_ACCESS_KEY_ID')
        self.access_key_secret = os.getenv('OSS_ACCESS_KEY_SECRET')
        self.bucket_name = os.getenv('OSS_BUCKET_NAME')
        self.endpoint = os.getenv('OSS_ENDPOINT')
        self.region = os.getenv('OSS_REGION', 'oss-cn-hangzhou')
        
        # 初始化 bucket
        auth = oss2.Auth(self.access_key_id, self.access_key_secret)
        self.bucket = oss2.Bucket(auth, self.endpoint, self.bucket_name)
    
    def generate_upload_url(self, filename: str, expire_seconds: int = 3600) -> Tuple[str, str]:
        """
        生成预签名上传URL
        
        Returns:
            (upload_url, file_key)
        """
        # 生成唯一文件 key
        ext = filename.rsplit('.', 1)[-1] if '.' in filename else ''
        file_key = f"uploads/{datetime.now().strftime('%Y%m%d')}/{uuid.uuid4().hex}.{ext}"
        
        # 生成预签名URL（客户端直接上传到OSS）
        upload_url = self.bucket.sign_url(
            'PUT',
            file_key,
            expire_seconds,
            headers={'Content-Type': 'application/octet-stream'}
        )
        
        return upload_url, file_key
    
    def generate_download_url(self, file_key: str, expire_seconds: int = 3600) -> str:
        """生成预签名下载URL"""
        return self.bucket.sign_url('GET', file_key, expire_seconds)
    
    def get_file_path(self, file_key: str) -> str:
        """获取文件的 OSS URL"""
        return f"https://{self.bucket_name}.{self.endpoint}/{file_key}"


oss_storage = OSSStorage()
```

```typescript
// frontend/src/api/upload.ts
import axios from 'axios'

/**
 * 获取预签名URL
 */
async function getPresignedUrl(filename: string) {
  const response = await axios.get('/api/oss/presigned-url', {
    params: { filename }
  })
  return response.data
}

/**
 * 直传文件到 OSS
 */
async function uploadToOSS(
  file: File,
  onProgress?: (percent: number) => void
) {
  // 1. 获取预签名URL
  const { upload_url, file_key } = await getPresignedUrl(file.name)
  
  // 2. 直接上传到 OSS
  await axios.put(upload_url, file, {
    headers: {
      'Content-Type': file.type || 'application/octet-stream'
    },
    onUploadProgress: (progressEvent) => {
      const percent = Math.round(
        (progressEvent.loaded * 100) / (progressEvent.total || file.size)
      )
      onProgress?.(percent)
    }
  })
  
  // 3. 返回 file_key（用于后续调用 Agent）
  return { file_key }
}

export { getPresignedUrl, uploadToOSS }
```
