    async function generateAuthUrl(host, path, apiKey, apiSecret) {
        const date = new Date().toUTCString(); // GMT 格式

            appendResponse(`  系统角色: ${systemRoleContent.substring(0, 100)}${systemRoleContent.length > 100 ? '...' : ''}`, 'info');
        }
        appendResponse(``, 'info');
       
        // 组装消息数组
        let messages = [];
           messages.push({
                role: "system",
                content: `
                你是一个面试专家，针对提交到简历做一个优化分析，给出优化建议和修改后的简历文本。
                1. 优化建议请详细说明，至少5点以上。
                2. 修改后的简历文本请保持原有内容的基础上进行修改，不要删除原有内容，除非内容完全不相关或者错误。
                3. 输出格式为JSON，包含optimization_suggestions和optimized_resume两个字段，optimization_suggestions为优化建议，optimized_resume为修改后的简历文本。
                `
            });
        if (systemRoleContent) {
            messages.push({
                role: "system",
                content: systemRoleContent
            });
        }
        messages.push({
            role: "user",
            content: userMessage
        });
        
        try {
            // 生成 WebSocket URL
            const wsUrl = await generateAuthUrl(host, fullPath, apiKey, apiSecret);
            appendResponse('🔗 WebSocket 连接中...', 'info');
            
            ws = new WebSocket(wsUrl);
            
            ws.onopen = () => {
                appendResponse('✅ 连接已建立，发送请求...', 'info');
                
                // 构建请求数据
                const requestData = {
                    header: {
                        app_id: appId,
                    },
                    parameter: {
                        chat: {
                            domain: domain,
                            temperature: 0.6,
                            max_tokens: 2048,
                            top_k: 4
                        }
                    },
                    payload: {
                        message: {
                            text: messages
                        }
                    }
                };
                
                ws.send(JSON.stringify(requestData));
                appendResponse('📤 请求已发送，等待响应...\n', 'info');
            };
            
            ws.onmessage = (event) => {
                const data = JSON.parse(event.data);
                const header = data.header;
                const payload = data.payload;
                
                if (header.code !== 0) {
                    appendResponse(`❌ 错误 [${header.code}]: ${header.message}`, 'error');
                    ws.close();
                    return;
                }
                
                // 处理普通文本消息
                if (payload && payload.choices && payload.choices.text) {
                    const textContent = payload.choices.text[0].content;
                    appendStreamText(textContent);
                }
                
                // 处理联网搜索信源
                if (payload && payload.plugins && payload.plugins.text) {
                    for (const plugin of payload.plugins.text) {
                        if (plugin.name === 'ifly_search' && plugin.content) {
                            try {
                                const searchResults = JSON.parse(plugin.content);
                                appendResponse('\n\n📚 参考信源：', 'info');
                                searchResults.forEach(ref => {
                                    appendResponse(`  ${ref.index}. ${ref.title} - ${ref.url}`, 'info');
                                });
                            } catch (e) {
                                console.error('解析搜索信源失败', e);
                            }
                        }
                    }
                }
                
                // 结束标志
                if (header.status === 2) {
                    appendResponse('\n\n✅ [会话结束]', 'info');
                    if (payload && payload.usage && payload.usage.text) {
                        const usage = payload.usage.text;
                        appendResponse(`📊 Token使用统计:`, 'token-usage');
                        appendResponse(`  prompt_tokens: ${usage.prompt_tokens}`, 'token-usage');
                        appendResponse(`  completion_tokens: ${usage.completion_tokens}`, 'token-usage');
                        appendResponse(`  total_tokens: ${usage.total_tokens}`, 'token-usage');
                    }
                    ws.close();
                }
            };
            
            ws.onerror = (error) => {
                appendResponse(`❌ WebSocket 错误: ${error.message || '未知错误'}`, 'error');
                console.error(error);
                isGenerating = false;
                sendBtn.disabled = false;
                stopBtn.disabled = true;
            };
            
            ws.onclose = () => {
                appendResponse('🔌 连接已关闭', 'info');
                ws = null;
                isGenerating = false;
                sendBtn.disabled = false;
                stopBtn.disabled = true;
            };
            
        } catch (err) {
            appendResponse(`❌ 发生异常: ${err.message}`, 'error');
            console.error(err);
            isGenerating = false;
            sendBtn.disabled = false;
            stopBtn.disabled = true;
        }
    });
